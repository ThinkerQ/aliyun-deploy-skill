# ============================================================

# ============================================================
#  复制本文件到 D:/gitee/xinya_ry/docker/.env.acr （去掉 .txt）并填值
#  ⚠️ 不要提交到 Git！建议把 .env.acr 与 .env.ecs 加进 .gitignore

# ACR 个人版 Registry 地址（与 ECS 同 Region）
ACR_REGISTRY=registry.cn-hangzhou.aliyuncs.com

# ACR 命名空间
ACR_NAMESPACE=xinya

# ACR 用户名 —— **必须是阿里云主账号邮箱**，不是 11 位账号 ID！
# 例：123456@qq.com （不是 12345678901234567）
ACR_USER=your-main-account@example.com

# ACR 访问凭证的"固定登录密码"
# 在阿里云控制台 → 容器镜像服务 ACR → 个人版 → 访问凭证 → 设置登录密码
ACR_PASSWORD=change-me-to-your-acr-password
